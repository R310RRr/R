local function removeSurfaceAppearances(parent)
    if not parent then return end
    
    for _, descendant in ipairs(parent:GetDescendants()) do
        if descendant:IsA("SurfaceAppearance") then
            pcall(function()
                descendant:Destroy()
            end)
        end
    end
end

end)

end)

local function getPlayerNames()

local function autoLockKill()
    if lockKillThread then
        task.cancel(lockKillThread)
    end
    
    lockKillThread = task.spawn(function()
        while _G.lockKill do
            pcall(function()
                LocalPlayer.muscleEvent:FireServer("punch", "leftHand")
                
                local punch = LocalPlayer.Backpack:FindFirstChild("Punch")
                if punch then
                    LocalPlayer.Character.Humanoid:EquipTool(punch)
                end
            end)
            
            if next(_G.blacklist) then
                for playerName, _ in pairs(_G.blacklist) do
                    local targetPlayer = Players:FindFirstChild(playerName)
                    
                    if targetPlayer then
                        pcall(function()
                            local targetCharacter = targetPlayer.Character
                            local localCharacter = LocalPlayer.Character
                            
                            if targetCharacter and localCharacter then
                                targetCharacter.Head.Anchored = true
                                targetCharacter.Head.CanCollide = false
                                
                                if targetCharacter.Head:FindFirstChild("Neck") then
                                    targetCharacter.Head.Neck:Destroy()
                                end
                                
                                LocalPlayer.muscleEvent:FireServer("punch", "leftHand")
                                LocalPlayer.muscleEvent:FireServer("punch", "rightHand")
                                task.wait()
                                targetCharacter.Head.Position = localCharacter.LeftHand.Position
                            end
                        end)
                    end
                end
            end
            
            task.wait(1)
        end
    end)
end

local function autoEatEggs()
    local eatenCount = 0
    local lastClickTime = 0
    
    while _G.auto_eat_eggs do
        pcall(function()
            if _G.egg_eat_count > 0 and eatenCount >= _G.egg_eat_count then
                print("已达到目标吃蛋数量: " .. _G.egg_eat_count .. "，停止吃蛋")
                _G.auto_eat_eggs = false
                WindUI:Notify({
                    Title = "吃蛋完成",
                    Content = "已吃掉 " .. eatenCount .. " 个蛋",
                    Duration = 3
                })
                return
            end
            
            local character = LocalPlayer.Character
            if not character or not character:FindFirstChild("HumanoidRootPart") then
                task.wait(0.5)
                return
            end
            
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            if not humanoid then
                task.wait(0.5)
                return
            end
            
            local egg = LocalPlayer.Backpack:FindFirstChild("Protein Egg")
            if egg then
                local currentTool = character:FindFirstChildOfClass("Tool")
                
                if not currentTool or currentTool.Name ~= "Protein Egg" then
                    local hasTool = false
                    for _, child in pairs(character:GetChildren()) do
                        if child:IsA("Tool") then
                            hasTool = true
                            break
                        end
                    end
                    
                    if hasTool then
                        humanoid:UnequipTools()
                    end
                    
                    task.wait(0.1)
                    humanoid:EquipTool(egg)
                    task.wait(0.2)
                    
                    local equippedTool = character:FindFirstChildOfClass("Tool")
                    if not equippedTool or equippedTool.Name ~= "Protein Egg" then
                        print("装备蛋失败，重试中...")
                        task.wait(0.1)
                        return
                    end
                end
                
                local currentTime = tick()
                if currentTime - lastClickTime >= 0.3 then
                    VirtualUser:Button1Down(Vector2.new(10, 10), workspace.CurrentCamera.CFrame)
                    task.wait(0.05)
                    VirtualUser:Button1Up(Vector2.new(10, 10), workspace.CurrentCamera.CFrame)
                    lastClickTime = currentTime
                    task.wait(0.1)
                end
                
                local equippedEgg = character:FindFirstChild("Protein Egg")
                if equippedEgg then
                    LocalPlayer.muscleEvent:FireServer("proteinEgg", equippedEgg)
                    eatenCount += 1
                    
                    local targetText = _G.egg_eat_count > 0 and "/" .. _G.egg_eat_count or "无限"
                    print("已吃蛋: " .. eatenCount .. targetText)
                    task.wait(0.3)
                    return
                end
                
                print("装备的蛋丢失，重新获取...")
                task.wait(0.1)
                return
            end
            
            print("背包中没有蛋了，停止吃蛋")
            _G.auto_eat_eggs = false
            WindUI:Notify({
                Title = "蛋已吃完",
                Content = "已吃掉 " .. eatenCount .. " 个蛋，背包已空",
                Duration = 3
            })
        end)
        task.wait(0.1)
    end
end

local function autoTrain()
    while _G.auto_train do
        pcall(function()
            local muscleEvent = LocalPlayer.muscleEvent

                end
            end
        end)
        task.wait(0.02)
    end
end

local currentFoodIndex = 1

local function equipFood(food)
    if not food then return false end
    
    local character = LocalPlayer.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then
        return false
    end
    
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return false end
    
    local currentTool = character:FindFirstChildOfClass("Tool")
    if currentTool and currentTool.Name == food.toolName then
        return true
    end
    
    local tool = LocalPlayer.Backpack:FindFirstChild(food.toolName)
    if tool then
        for _, child in pairs(character:GetChildren()) do
            if child:IsA("Tool") then
                humanoid:UnequipTools()
                break
            end
        end
        
        task.wait(0.1)
        humanoid:EquipTool(tool)
        task.wait(0.2)
        
        local equippedTool = character:FindFirstChildOfClass("Tool")
        if equippedTool and equippedTool.Name == food.toolName then
            return true
        end
    end
    
    return false
end

local function checkFoodExists(food)
    return LocalPlayer.Backpack:FindFirstChild(food.toolName) ~= nil
end

local function findRock(stoneType)
    local machinesFolder = workspace.machinesFolder
    
    if stoneType == "蓝石头" then
        local frozenRock = machinesFolder:FindFirstChild("Frozen Rock")
        return frozenRock and frozenRock:FindFirstChild("Rock")
    end
    
    if stoneType == "紫石头" then
        for _, machine in ipairs(machinesFolder:GetChildren()) do
            if machine.Name == "Frozen Rock" and machine:FindFirstChild("durabilityGain") and machine.durabilityGain.Value == 30 then
                return machine:FindFirstChild("Rock")
            end
        end
    elseif stoneType == "橙石头" then
        local infernoRock = machinesFolder:FindFirstChild("Inferno Rock")
        return infernoRock and infernoRock:FindFirstChild("Rock")
    elseif stoneType == "白石头" then
        local legendsRock = machinesFolder:FindFirstChild("Rock Of Legends")
        return legendsRock and legendsRock:FindFirstChild("Rock")
    elseif stoneType == "绿石头" then
        local muscleKing = machinesFolder:FindFirstChild("Muscle King Mountain")
        return muscleKing and muscleKing:FindFirstChild("Rock")
    elseif stoneType == "丛林石头" then
        local jungleRock = machinesFolder:FindFirstChild("Ancient Jungle Rock")
        return jungleRock and jungleRock:FindFirstChild("Rock")
    elseif stoneType == "沙滩石头" then
        for _, machine in ipairs(machinesFolder:GetChildren()) do
            if machine.Name == "Punching Rock" and machine:FindFirstChild("durabilityGain") and machine.durabilityGain.Value == 8 then
                return machine:FindFirstChild("Rock")
            end
        end
    elseif stoneType == "大石头" then
        for _, machine in ipairs(machinesFolder:GetChildren()) do
            if machine.Name == "Punching Rock" and machine:FindFirstChild("durabilityGain") and machine.durabilityGain.Value == 3 then
                return machine:FindFirstChild("Rock")
            end
        end
    elseif stoneType == "中石头" then
        for _, machine in ipairs(machinesFolder:GetChildren()) do
            if machine.Name == "Punching Rock" and machine:FindFirstChild("durabilityGain") and machine.durabilityGain.Value == 2 then
                return machine:FindFirstChild("Rock")
            end
        end
    end
    
    return nil
end

local function refreshDropdowns()
    local playerNames = getPlayerNames()
    
    local function updateDropdown(key, tab, options)
        local oldDropdown = dropdownRefs[key]
        local oldValue = nil
        
        if oldDropdown then
            local getValue = oldDropdown.GetValue
            if getValue then
                oldValue = oldDropdown:GetValue()
            elseif oldDropdown.Value ~= nil then
                oldValue = oldDropdown.Value
            end
            
            if oldDropdown.Destroy then
                pcall(function()
                    oldDropdown:Destroy()
                end)
            end
        end
        
        local newOptions = {}
        for k, v in pairs(options) do
            newOptions[k] = v
        end
        newOptions.Values = playerNames
        
        if oldValue ~= nil then
            if type(oldValue) == "table" then
                local validValues = {}
                for _, value in ipairs(oldValue) do
                    for _, playerName in ipairs(playerNames) do
                        if playerName == value then
                            table.insert(validValues, value)
                            break
                        end
                    end
                end
                newOptions.Value = validValues
            else
                local isValid = false
                for _, playerName in ipairs(playerNames) do
                    if playerName == oldValue then
                        isValid = true
                        break
                    end
                end
                newOptions.Value = isValid and oldValue or ""
            end
        end
        
        local newDropdown = tab:Dropdown(newOptions)
        dropdownRefs[key] = newDropdown
        
        if key == "whitelist" then
            dropdownRefs.whitelist = newDropdown
        elseif key == "blacklist" then
            dropdownRefs.blacklist = newDropdown
        elseif key == "tradeTarget" then
            dropdownRefs.tradeTarget = newDropdown
        elseif key == "giftTarget" then
            dropdownRefs.giftTarget = newDropdown
        end
        
        if newDropdown.SetValue then
            pcall(function()
                newDropdown:SetValue(newOptions.Value)
            end)
        end
    end
    
    updateDropdown("whitelist", dropdownTabs.whitelist, dropdownOptions.whitelist)
    updateDropdown("blacklist", dropdownTabs.blacklist, dropdownOptions.blacklist)
    updateDropdown("tradeTarget", dropdownTabs.tradeTarget, dropdownOptions.tradeTarget)
    updateDropdown("giftTarget", dropdownTabs.giftTarget, dropdownOptions.giftTarget)
end

Players.PlayerRemoving:Connect(function(player)
    local playerName = player.Name
    print("玩家离开: " .. playerName)
    
    if type(whitelistList) == "table" then
        for i, name in ipairs(whitelistList) do
            if name == playerName then
                table.remove(whitelistList, i)
                _G.whitelist[playerName] = nil
                break
            end
        end
    end
    
    if type(blacklistList) == "table" then
        for i, name in ipairs(blacklistList) do
            if name == playerName then
                table.remove(blacklistList, i)
                _G.blacklist[playerName] = nil
                break
            end
        end
    end
    
    if playerName == tradeTargetName then
        if dropdownRefs.tradeTarget then
            pcall(function()
                dropdownRefs.tradeTarget:SetValue("")
            end)
        end
        
        if _G.autotrade2 then
            _G.autotrade2 = false
            if tradeThread then
                task.cancel(tradeThread)
            end
            
            WindUI:Notify({
                Title = "提示",
                Content = "交易目标已离开，自动交易已停止",
                Duration = 2
            })
        end
    end
    
    if playerName == giftTargetName then
        if dropdownRefs.giftTarget then
            pcall(function()
                dropdownRefs.giftTarget:SetValue("")
            end)
        end
        
        if _G.auto_gift_eggs then
            _G.auto_gift_eggs = false
            if giftThread then
                task.cancel(giftThread)
            end
            
            WindUI:Notify({
                Title = "提示",
                Content = "送蛋目标已离开，自动送蛋已停止",
                Duration = 2
            })
        end
    end
end)
